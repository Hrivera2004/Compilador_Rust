fn max(a: i32, b: i32) -> i32 {
    if a >= b {
        return a;
    } else {
        return b;
    }
}

/* Suma los numeros del rango [0, n)
   usando un ciclo for */
fn suma(n: i32) -> i32 {
    let total: i32 = 0;
    for i in 0..n {
        total = total + i;
    }
    return total;
}
